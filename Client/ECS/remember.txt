package ECS

import "core:math"


// ============================================================
// ADD SPHERE TERRAIN
// ============================================================

add_sphere_terrain :: proc(
    world: ^World_Grid,
    position: [3]f64,
    radius: f64,
    value: World_Grid_Value,
) {
    if radius <= 0 {
        return
    }

    // The position is already in the coordinate space
    // used by this region's octree.
    add_sphere_to_node(
        &world.tree.root,
        position,
        radius,
        value,
        0,
        [3]f64{0, 0, 0},
        f64(WORLD_REGION_SIZE) * 0.5,
    )
}


// ============================================================
// RECURSIVE SPHERE EDIT
// ============================================================

add_sphere_to_node :: proc(
    node: ^World_Grid_Node,
    sphere_center: [3]f64,
    radius: f64,
    value: World_Grid_Value,
    depth: int,
    node_center: [3]f64,
    half_size: f64,
) {
    // The sphere does not touch this node at all.
    if !sphere_intersects_box(
        sphere_center,
        radius,
        node_center,
        half_size,
    ) {
        return
    }

    // The sphere completely covers this node.
    // The entire node can therefore become uniform.
    if sphere_contains_box(
        sphere_center,
        radius,
        node_center,
        half_size,
    ) {
        set_node_uniform(node, value)
        return
    }

    // We reached the smallest possible node.
    if depth >= WORLD_GRID_MAX_DEPTH {
        set_node_leaf(node, value)
        return
    }

    // The sphere only partially covers this node.
    // Subdivide and let each child determine whether
    // it is inside, outside, or partially intersecting.
    subdivide_node(node)

    child_half_size := half_size * 0.5

    for i in 0..<WORLD_GRID_CHILD_COUNT {
        child_center := child_node_center(
            node_center,
            child_half_size,
            i,
        )

        add_sphere_to_node(
            &node.children[i],
            sphere_center,
            radius,
            value,
            depth + 1,
            child_center,
            child_half_size,
        )
    }

    // If all children became identical, the subdivision
    // is no longer necessary.
    collapse_node_if_uniform(node)
}


// ============================================================
// SUBDIVIDE NODE
// ============================================================

subdivide_node :: proc(
    node: ^World_Grid_Node,
) {
    if node.state == .Leaf {
        return
    }

    if len(node.children) != 0 {
        return
    }

    node.children = make(
        [dynamic]World_Grid_Node,
        WORLD_GRID_CHILD_COUNT,
    )

    // Every child initially contains the same value
    // as its parent.
    for i in 0..<WORLD_GRID_CHILD_COUNT {
        node.children[i] = World_Grid_Node{
            state = .Uniform,
            value = node.value,
        }
    }

    node.state = .Mixed
}


// ============================================================
// SET UNIFORM
// ============================================================

set_node_uniform :: proc(
    node: ^World_Grid_Node,
    value: World_Grid_Value,
) {
    delete_node_children(node)

    node.state = .Uniform
    node.value = value
}


// ============================================================
// SET LEAF
// ============================================================

set_node_leaf :: proc(
    node: ^World_Grid_Node,
    value: World_Grid_Value,
) {
    delete_node_children(node)

    node.state = .Leaf
    node.value = value
}


// ============================================================
// COLLAPSE NODE
// ============================================================

collapse_node_if_uniform :: proc(
    node: ^World_Grid_Node,
) {
    if len(node.children) != WORLD_GRID_CHILD_COUNT {
        return
    }

    first := &node.children[0]

    // A mixed child means this node cannot be uniform.
    if first.state == .Mixed {
        return
    }

    first_value := first.value

    for i in 1..<WORLD_GRID_CHILD_COUNT {
        child := &node.children[i]

        if child.state == .Mixed {
            return
        }

        if !world_grid_values_equal(
            child.value,
            first_value,
        ) {
            return
        }
    }

    set_node_uniform(node, first_value)
}


// ============================================================
// VALUE COMPARISON
// ============================================================

world_grid_values_equal :: proc(
    a: World_Grid_Value,
    b: World_Grid_Value,
) -> bool {
    return a.density == b.density &&
           a.material == b.material
}


// ============================================================
// CHILD CENTER
// ============================================================

child_node_center :: proc(
    parent_center: [3]f64,
    child_half_size: f64,
    child_index: int,
) -> [3]f64 {
    center := parent_center

    if (child_index & 1) != 0 {
        center[0] += child_half_size
    } else {
        center[0] -= child_half_size
    }

    if (child_index & 2) != 0 {
        center[1] += child_half_size
    } else {
        center[1] -= child_half_size
    }

    if (child_index & 4) != 0 {
        center[2] += child_half_size
    } else {
        center[2] -= child_half_size
    }

    return center
}


// ============================================================
// SPHERE / BOX INTERSECTION
// ============================================================

sphere_intersects_box :: proc(
    sphere_center: [3]f64,
    radius: f64,
    box_center: [3]f64,
    half_size: f64,
) -> bool {
    dx := abs_f64(sphere_center[0] - box_center[0])
    dy := abs_f64(sphere_center[1] - box_center[1])
    dz := abs_f64(sphere_center[2] - box_center[2])

    closest_x := max_f64(dx - half_size, 0)
    closest_y := max_f64(dy - half_size, 0)
    closest_z := max_f64(dz - half_size, 0)

    distance_squared :=
        closest_x * closest_x +
        closest_y * closest_y +
        closest_z * closest_z

    return distance_squared <= radius * radius
}


// ============================================================
// SPHERE / BOX CONTAINMENT
// ============================================================

sphere_contains_box :: proc(
    sphere_center: [3]f64,
    radius: f64,
    box_center: [3]f64,
    half_size: f64,
) -> bool {
    dx := abs_f64(sphere_center[0] - box_center[0]) + half_size
    dy := abs_f64(sphere_center[1] - box_center[1]) + half_size
    dz := abs_f64(sphere_center[2] - box_center[2]) + half_size

    distance_squared :=
        dx * dx +
        dy * dy +
        dz * dz

    return distance_squared <= radius * radius
}


// ============================================================
// MATH HELPERS
// ============================================================

abs_f64 :: proc(x: f64) -> f64 {
    return math.abs(x)
}


max_f64 :: proc(a, b: f64) -> f64 {
    if a > b {
        return a
    }

    return b
}


// ============================================================
// DELETE CHILDREN
// ============================================================

delete_node_children :: proc(
    node: ^World_Grid_Node,
) {
    for i := 0; i < len(node.children); i += 1 {
        delete_node_children(&node.children[i])
    }

    delete(node.children)
    node.children = nil
}
